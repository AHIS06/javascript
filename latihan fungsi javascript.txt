const readline = require('readline');

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

// Fungsi void untuk mencetak identitas di paling atas
function cetakIdentitas() {
    console.log("---------------------");
    console.log("|     Harits        |");
    console.log("|     pasming       |");
    console.log("---------------------");
}

// 1. Fungsi Penjumlahan
function tambah(a = 10, b = 5) {
    return a + b;
}

// 2. Fungsi Pengurangan
function kurang(a = 10, b = 5) {
    return a - b;
}

// 3. Fungsi Perkalian
function kali(a = 10, b = 5) {
    return a * b;
}

// 4. Fungsi Pembagian
function bagi(a = 10, b = 5) {
    return b !== 0 ? a / b : "Tidak terdefinisi (pembagian dengan nol)";
}

// 5. Fungsi Modulus
function modulus(a = 10, b = 5) {
    return b !== 0 ? a % b : "Tidak terdefinisi (modulus dengan nol)";
}

// Fungsi untuk menjalankan program secara berulang
function mulaiProgram() {
    rl.question("Masukkan Menu(+|-|/|*|%|stop): ", (menu) => {
        let pilihan = menu.trim().toLowerCase();

        if (pilihan === 'stop') {
            console.log("Program Berhenti, Terima Kasih telah menggunakan program saya.");
            rl.close();
            return;
        }

        if (!['+', '-', '*', '/', '%'].includes(pilihan)) {
            console.log("Menu tidak valid, silakan coba lagi.\n");
            return mulaiProgram();
        }

        rl.question("Masukkan Nilai 1: ", (input1) => {
            rl.question("Masukkan Nilai 2: ", (input2) => {
                let nilai1 = input1.trim() === "" ? undefined : Number(input1);
                let nilai2 = input2.trim() === "" ? undefined : Number(input2);

                let hasil;
                let namaOperasi = "";

                // Menentukan fungsi mana yang akan dipanggil berdasarkan input user
                if (pilihan === '+') {
                    hasil = tambah(nilai1, nilai2);
                    namaOperasi = "Penjumlahan";
                } else if (pilihan === '-') {
                    hasil = kurang(nilai1, nilai2);
                    namaOperasi = "Pengurangan";
                } else if (pilihan === '*') {
                    hasil = kali(nilai1, nilai2);
                    namaOperasi = "Perkalian";
                } else if (pilihan === '/') {
                    hasil = bagi(nilai1, nilai2);
                    namaOperasi = "Pembagian";
                } else if (pilihan === '%') {
                    hasil = modulus(nilai1, nilai2);
                    namaOperasi = "Modulus";
                }

                // Mengambil nilai aktual yang digunakan untuk ditampilkan pada console.log
                let aAktual = (input1.trim() === "") ? 10 : nilai1;
                let bAktual = (input2.trim() === "") ? 5 : nilai2;

                console.log(`Hasil ${namaOperasi} ${aAktual} ${pilihan} ${bAktual} adalah ${hasil}\n`);

                // Mengulang program kembali
                mulaiProgram();
            });
        });
    });
}

// Eksekusi Program
cetakIdentitas();
mulaiProgram();
